# Сборка проекта с нуля

```
pnpm create vite@latest PROJECT_NAME --template vue-ts

pnpm add -D @types/node @types/lodash-es cross-env

// essential libs
pnpm add @vueuse/components @vueuse/core dayjs lodash-es maska pinia pinia-plugin-persistedstate vue-i18n vue-router @phosphor-icons/vue

pnpm add tailwindcss @tailwindcss/vite

pnpm dlx shadcn-vue@latest init
```
